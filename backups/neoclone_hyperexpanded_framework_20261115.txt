    # ========== ORIGINAL 10 DIMENSIONS ==========

    def _classify_vol_regime(self) -> pd.Series:
        """Volatility regime: FAST_SMOOTH, FAST_ERRATIC, SLOW_SMOOTH, SLOW_ERRATIC"""
        atr = self._compute_atr(period=14)
        slope = self._compute_slope(self.df['close'], period=3)
        slope_std = slope.rolling(5).std()

        regimes = []
        for i in range(len(self.df)):
            vol_high = atr.iloc[i] > atr.mean() * 1.2
            slope_smooth = slope_std.iloc[i] < slope_std.mean() * 0.8

            if vol_high and slope_smooth:
                regime = 'FAST_SMOOTH'
            elif vol_high and not slope_smooth:
                regime = 'FAST_ERRATIC'
            elif not vol_high and slope_smooth:
                regime = 'SLOW_SMOOTH'
            else:
                regime = 'SLOW_ERRATIC'

            regimes.append(regime)

        return pd.Series(regimes, index=self.df.index)

    def _detect_spikes(self) -> List[Dict]:
        """Detect price spikes: magnitude > 2 sigma"""
        returns = self.df['close'].pct_change()
        mean_return = returns.mean()
        std_return = returns.std()
        threshold = 2.0

        spikes = []
        for i in range(len(self.df)):
            if abs(returns.iloc[i] - mean_return) > threshold * std_return:
                spikes.append({
                    'idx': i,
                    'time': self.df.index[i],
                    'magnitude': returns.iloc[i],
                    'direction': 'UP' if returns.iloc[i] > 0 else 'DOWN'
                })

        return spikes

    def _detect_slope_changes(self) -> List[Dict]:
        """Detect trend direction changes"""
        slope = self._compute_slope(self.df['close'], period=5)
        changes = []

        for i in range(1, len(slope)):
            if (slope.iloc[i-1] > 0 and slope.iloc[i] < 0) or \
               (slope.iloc[i-1] < 0 and slope.iloc[i] > 0):
                changes.append({
                    'idx': i,
                    'time': self.df.index[i],
                    'from': 'UP' if slope.iloc[i-1] > 0 else 'DOWN',
                    'to': 'DOWN' if slope.iloc[i] < 0 else 'UP',
                    'magnitude': abs(slope.iloc[i] - slope.iloc[i-1])
                })

        return changes

    def _extract_time_patterns(self) -> Dict:
        """Extract hour-of-day, day-of-week, session patterns"""
        time_data = {
            'hour_of_day': self.df.index.hour.value_counts().to_dict(),
            'day_of_week': self.df.index.dayofweek.value_counts().to_dict(),
            'sessions': self._classify_sessions()
        }
        return time_data

    def _classify_sessions(self) -> Dict:
        """Classify as Asian/London/NY session based on hour UTC"""
        sessions = {'ASIAN': 0, 'LONDON': 0, 'NY': 0}
        for hour in self.df.index.hour:
            if 0 <= hour < 8:
                sessions['ASIAN'] += 1
            elif 8 <= hour < 16:
                sessions['LONDON'] += 1
            else:
                sessions['NY'] += 1
        return sessions

    def _compute_sma_suite(self) -> Dict:
        """Compute comprehensive SMA range: 25 to 500+ """
        sma_periods = [25, 50, 100, 150, 200, 250, 300, 350, 400, 450, 500]
        if self.tf in ['H1', 'D1']:
            sma_periods.extend([750, 1000])

        smas = {}
        for period in sma_periods:
            if period <= len(self.df):
                smas[period] = self.df['close'].rolling(period).mean()

        return smas

    def _detect_crossovers(self) -> List[Dict]:
        """Detect all SMA crossovers (golden/death crosses)"""
        sma_suite = self.features['sma_suite']
        crossovers = []

        sma_list = sorted(sma_suite.keys())
        for i in range(len(sma_list)-1):
            fast_period = sma_list[i]
            slow_period = sma_list[i+1]
            fast_sma = sma_suite[fast_period]
            slow_sma = sma_suite[slow_period]

            for j in range(1, len(self.df)):
                if fast_sma.iloc[j-1] <= slow_sma.iloc[j-1] and fast_sma.iloc[j] > slow_sma.iloc[j]:
                    crossovers.append({
                        'idx': j, 'time': self.df.index[j],
                        'type': 'GOLDEN',
                        'fast_sma': fast_period,
                        'slow_sma': slow_period,
                        'price': self.df['close'].iloc[j]
                    })
                elif fast_sma.iloc[j-1] >= slow_sma.iloc[j-1] and fast_sma.iloc[j] < slow_sma.iloc[j]:
                    crossovers.append({
                        'idx': j, 'time': self.df.index[j],
                        'type': 'DEATH',
                        'fast_sma': fast_period,
                        'slow_sma': slow_period,
                        'price': self.df['close'].iloc[j]
                    })

        return crossovers

    def _compute_convergence(self) -> Dict:
        """Compute SMA convergence/divergence/overlap"""
        sma_suite = self.features['sma_suite']
        convergence_stats = {
            'avg_spread': [],
            'convergence_periods': [],
            'divergence_periods': []
        }

        for i in range(len(self.df)):
            sma_values = [sma.iloc[i] for sma in sma_suite.values() if not pd.isna(sma.iloc[i])]
            if len(sma_values) > 1:
                spread = max(sma_values) - min(sma_values)
                convergence_stats['avg_spread'].append(spread)

                if spread < np.percentile(convergence_stats['avg_spread'], 25):
                    convergence_stats['convergence_periods'].append(i)
                elif spread > np.percentile(convergence_stats['avg_spread'], 75):
                    convergence_stats['divergence_periods'].append(i)

        return convergence_stats

    def _compute_fib_levels(self) -> Dict:
        """Compute Fibonacci retracement levels"""
        recent_high = self.df['high'].rolling(50).max().iloc[-1]
        recent_low = self.df['low'].rolling(50).min().iloc[-1]
        diff = recent_high - recent_low

        fibs = {
            '0.0': recent_low,
            '0.236': recent_low + 0.236 * diff,
            '0.382': recent_low + 0.382 * diff,
            '0.618': recent_low + 0.618 * diff,
            '1.0': recent_high
        }
        return fibs

    def _compute_sr_zones(self) -> Dict:
        """Identify support/resistance zones"""
        highs = self.df['high'].rolling(10).max()
        lows = self.df['low'].rolling(10).min()

        sr_zones = {
            'resistance': highs.unique().tolist()[-5:],
            'support': lows.unique().tolist()[-5:]
        }
        return sr_zones

    def _compute_psych_levels(self) -> List:
        """Identify psychological levels (round numbers)"""
        current_price = self.df['close'].iloc[-1]
        psych_levels = []

        magnitude = 10 ** np.floor(np.log10(current_price))
        for multiplier in range(int(current_price / magnitude) - 5,
                                int(current_price / magnitude) + 10):
            psych_level = multiplier * magnitude
            if abs(psych_level - current_price) / current_price < 0.05:
                psych_levels.append(psych_level)

        return sorted(psych_levels)

    def _detect_overbought_oversold(self) -> Dict:
        """Detect overbought/oversold using RSI and mean overstretch"""
        rsi = self._compute_rsi(period=14)
        mean_price = self.df['close'].rolling(20).mean()
        overstretch = []

        overbought = []
        oversold = []

        for i in range(len(rsi)):
            if rsi.iloc[i] > 70:
                overbought.append({'idx': i, 'time': self.df.index[i], 'rsi': rsi.iloc[i]})
            elif rsi.iloc[i] < 30:
                oversold.append({'idx': i, 'time': self.df.index[i], 'rsi': rsi.iloc[i]})

            if not pd.isna(mean_price.iloc[i]):
                deviation = (self.df['close'].iloc[i] - mean_price.iloc[i]) / mean_price.iloc[i]
                if abs(deviation) > 0.03:
                    overstretch.append({
                        'idx': i,
                        'time': self.df.index[i],
                        'deviation': deviation,
                        'direction': 'UP' if deviation > 0 else 'DOWN'
                    })

        return {
            'overbought': overbought,
            'oversold': oversold,
            'mean_overstretch': overstretch
        }

    def _detect_hype_correction_waves(self) -> List[Dict]:
        """Detect hype waves (sharp moves) vs correction waves (reversal moves)"""
        returns = self.df['close'].pct_change()
        volatility = returns.rolling(10).std()
        waves = []

        for i in range(1, len(self.df)):
            if volatility.iloc[i] > volatility.mean() * 1.5:
                if abs(returns.iloc[i]) > returns.std() * 1.5:
                    wave_type = 'HYPE_WAVE' if i > 1 and returns.iloc[i] * returns.iloc[i-1] > 0 else 'CORRECTION_WAVE'
                    waves.append({
                        'idx': i,
                        'time': self.df.index[i],
                        'type': wave_type,
                        'magnitude': returns.iloc[i],
                        'velocity': abs(returns.iloc[i] / (volatility.iloc[i] + 1e-6))
                    })

        return waves

    def _classify_reversal_drivers(self) -> List[Dict]:
        """Classify what drives reversals: technical, correction, rumor, news, speculation"""
        reversals = []
        slope = self._compute_slope(self.df['close'], period=5)

        for i in range(1, len(slope)):
            if (slope.iloc[i-1] > 0 and slope.iloc[i] < 0) or (slope.iloc[i-1] < 0 and slope.iloc[i] > 0):
                rsi = self._compute_rsi(period=14).iloc[i]

                if rsi > 70 or rsi < 30:
                    driver = 'TECHNICAL_OVERBOUGHT'
                elif abs(self.df['close'].iloc[i] - self.df['close'].iloc[max(0, i-10)]) / self.df['close'].iloc[max(0, i-10)] > 0.02:
                    driver = 'CORRECTION_MOVE'
                else:
                    driver = 'RUMOR_OR_SPECULATION'

                reversals.append({
                    'idx': i,
                    'time': self.df.index[i],
                    'driver': driver,
                    'rsi': rsi,
                    'magnitude': abs(slope.iloc[i] - slope.iloc[i-1])
                })

        return reversals

    def _detect_correction_quality(self) -> Dict:
        """Detect correction vs over-correction per Nasif's specs:
        Normal correction: 30-50% depth, 5-15 bars
        Over-correction: >70% depth, 2-5 bars, sharp velocity
        """
        peaks = []
        troughs = []
        close = self.df['close'].values

        for i in range(1, len(close) - 1):
            if close[i] > close[i-1] and close[i] > close[i+1]:
                peaks.append(i)
            elif close[i] < close[i-1] and close[i] < close[i+1]:
                troughs.append(i)

        corrections = {
            'normal_corrections': [],
            'overcorrections': [],
            'undefined': []
        }

        for p in peaks:
            next_troughs = [t for t in troughs if p < t < p + 30]
            if not next_troughs:
                continue

            for t in next_troughs:
                correction_depth = (close[p] - close[t]) / close[p]
                bar_duration = t - p
                velocity = abs(correction_depth / max(bar_duration, 1))

                if 0.30 <= correction_depth <= 0.50 and 5 <= bar_duration <= 15:
                    corrections['normal_corrections'].append({
                        'idx': t,
                        'time': self.df.index[t],
                        'depth': correction_depth,
                        'duration': bar_duration,
                        'velocity': velocity
                    })
                elif correction_depth > 0.70 and 2 <= bar_duration <= 5 and velocity > 0.1:
                    corrections['overcorrections'].append({
                        'idx': t,
                        'time': self.df.index[t],
                        'depth': correction_depth,
                        'duration': bar_duration,
                        'velocity': velocity
                    })
                else:
                    corrections['undefined'].append({
                        'idx': t,
                        'depth': correction_depth,
                        'duration': bar_duration
                    })

        return corrections

    def _test_formula_relationships(self) -> Dict:
        """Test linear, quadratic, cubic, geometric formula fits on slope/trend"""
        slope = self._compute_slope(self.df['close'], period=5)
        x = np.arange(len(slope))

        results = {}
        valid_idx = ~pd.isna(slope)

        if valid_idx.sum() > 10:
            y = slope[valid_idx].values
            x_valid = x[valid_idx]

            # Linear: y = ax + b
            z_linear = np.polyfit(x_valid, y, 1)
            results['linear'] = {'coeff': z_linear.tolist(), 'r2': np.corrcoef(x_valid, y)[0,1]**2}

            # Quadratic: y = ax^2 + bx + c
            z_quad = np.polyfit(x_valid, y, 2)
            results['quadratic'] = {'coeff': z_quad.tolist(), 'r2': np.corrcoef(x_valid, y**2)[0,1]**2}

            # Cubic: y = ax^3 + bx^2 + cx + d
            z_cubic = np.polyfit(x_valid, y, 3)
            results['cubic'] = {'coeff': z_cubic.tolist(), 'r2': np.corrcoef(x_valid, y**3 if (y>0).all() else y)[0,1]**2}

            # Geometric (log fit): log(y) = ax + b
            try:
                y_pos = np.abs(y) + 1e-6
                z_geom = np.polyfit(x_valid, np.log(y_pos), 1)
                results['geometric'] = {'coeff': z_geom.tolist(), 'r2': 0.5}
            except:
                results['geometric'] = {'coeff': [], 'r2': 0}

        return results

    def _analyze_volatility_persistence(self) -> Dict:
        """Analyze session-to-session volatility carryover (especially London)"""
        if len(self.df) < 48:  # Need multiple days
            return {'sessions': [], 'persistence_score': 0}

        daily_vol = {}
        for date in self.df.index.date:
            day_data = self.df[self.df.index.date == date]
            if len(day_data) > 0:
                vol = day_data['close'].std() / day_data['close'].mean()
                daily_vol[date] = vol

        persistence = []
        dates = sorted(daily_vol.keys())
        for i in range(1, len(dates)):
            if i > 0:
                corr = daily_vol[dates[i]] / (daily_vol[dates[i-1]] + 1e-6)
                persistence.append({
                    'from_date': dates[i-1],
                    'to_date': dates[i],
                    'carryover_ratio': corr
                })

        persistence_score = np.mean([p['carryover_ratio'] for p in persistence]) if persistence else 0

        return {
            'daily_volatility': {str(k): v for k, v in daily_vol.items()},
            'session_persistence': persistence,
            'persistence_score': persistence_score
        }

    def _recognize_chart_patterns(self) -> List[Dict]:
        """Recognize chart patterns: peaks/valleys, diagonals, triangles, head-and-shoulders"""
        patterns = []

        # Detect peaks and valleys
        highs = self.df['high'].rolling(window=5, center=True).max()
        lows = self.df['low'].rolling(window=5, center=True).min()

        for i in range(2, len(self.df) - 2):
            if self.df['high'].iloc[i] == highs.iloc[i] and highs.iloc[i] > highs.iloc[i-1] and highs.iloc[i] > highs.iloc[i+1]:
                patterns.append({'idx': i, 'time': self.df.index[i], 'type': 'PEAK', 'price': self.df['high'].iloc[i]})

            if self.df['low'].iloc[i] == lows.iloc[i] and lows.iloc[i] < lows.iloc[i-1] and lows.iloc[i] < lows.iloc[i+1]:
                patterns.append({'idx': i, 'time': self.df.index[i], 'type': 'VALLEY', 'price': self.df['low'].iloc[i]})

        return patterns

    def _detect_mean_reversion(self) -> Dict:
        """Detect mean reversion strength and scoring"""
        mean_price = self.df['close'].rolling(30).mean()
        deviation = (self.df['close'] - mean_price) / mean_price

        reversion_strength = []
        for i in range(1, len(deviation)):
            if (deviation.iloc[i-1] > 0.02 and deviation.iloc[i] < deviation.iloc[i-1]) or \
               (deviation.iloc[i-1] < -0.02 and deviation.iloc[i] > deviation.iloc[i-1]):
                reversion_strength.append({
                    'idx': i,
                    'time': self.df.index[i],
                    'deviation': deviation.iloc[i-1],
                    'reversion_rate': abs(deviation.iloc[i] - deviation.iloc[i-1])
                })

        return {
            'mean_reversion_events': reversion_strength,
            'avg_reversion_rate': np.mean([r['reversion_rate'] for r in reversion_strength]) if reversion_strength else 0
        }

    def _compute_volatility_clustering(self) -> Dict:
        """Compute volatility clustering (GARCH-like) and regime persistence"""
        returns = self.df['close'].pct_change()
        vol = returns.rolling(10).std()

        clustering = []
        for i in range(1, len(vol)):
            if vol.iloc[i] > vol.iloc[i-1] * 1.1:
                clustering.append({
                    'idx': i,
                    'time': self.df.index[i],
                    'vol_spike': vol.iloc[i] / vol.iloc[i-1],
                    'persistence': 'HIGH' if i + 1 < len(vol) and vol.iloc[i+1] > vol.iloc[i] * 0.9 else 'MEAN_REVERTING'
                })

        return {
            'vol_clusters': clustering,
            'clustering_score': len(clustering) / len(vol) if len(vol) > 0 else 0
        }

    def _detect_reverse_reversals(self) -> List[Dict]:
        """Detect reverse reversals (fake-outs, bull/bear traps)"""
        slope = self._compute_slope(self.df['close'], period=5)
        fake_reversals = []

        for i in range(2, len(slope) - 1):
            if (slope.iloc[i-1] > 0 and slope.iloc[i] < 0 and slope.iloc[i+1] > 0) or \
               (slope.iloc[i-1] < 0 and slope.iloc[i] > 0 and slope.iloc[i+1] < 0):
                fake_reversals.append({
                    'idx': i,
                    'time': self.df.index[i],
                    'type': 'BULL_TRAP' if slope.iloc[i] > 0 else 'BEAR_TRAP',
                    'reversal_magnitude': abs(slope.iloc[i])
                })

        return fake_reversals

    def _score_intuition_heuristics(self) -> Dict:
        """Score pattern confidence using multiple heuristic signals"""
        rsi = self._compute_rsi(14)
        slope = self._compute_slope(self.df['close'], period=5)
        vol = self.df['close'].pct_change().rolling(10).std()

        confidence_scores = []
        for i in range(10, len(self.df)):
            signal_count = 0
            total_signals = 5

            if 40 < rsi.iloc[i] < 60:
                signal_count += 1
            if abs(slope.iloc[i]) > slope.std() * 0.5:
                signal_count += 1
            if vol.iloc[i] > vol.mean() * 0.8:
                signal_count += 1
            if self.df['volume'].iloc[i] > self.df['volume'].mean():
                signal_count += 1
            if (self.df['close'].iloc[i] - self.df['open'].iloc[i]) > (self.df['high'].iloc[i] - self.df['low'].iloc[i]) * 0.5:
                signal_count += 1

            confidence = signal_count / total_signals
            confidence_scores.append({
                'idx': i,
                'time': self.df.index[i],
                'confidence': confidence,
                'signal_count': signal_count
            })

        return {
            'confidence_scores': confidence_scores,
            'avg_confidence': np.mean([c['confidence'] for c in confidence_scores]) if confidence_scores else 0
        }

    # ========== HELPER METHODS ==========

    def _compute_rsi(self, period=14) -> pd.Series:
        """Compute RSI (Relative Strength Index)"""
        delta = self.df['close'].diff()
        gain = (delta.where(delta > 0, 0)).rolling(window=period).mean()
        loss = (-delta.where(delta < 0, 0)).rolling(window=period).mean()
        rs = gain / loss
        rsi = 100 - (100 / (1 + rs))
        return rsi

    def _compute_atr(self, period=14) -> pd.Series:
        """Compute Average True Range"""
        high_low = self.df['high'] - self.df['low']
        high_close = abs(self.df['high'] - self.df['close'].shift())
        low_close = abs(self.df['low'] - self.df['close'].shift())
        ranges = pd.concat([high_low, high_close, low_close], axis=1)
        true_range = ranges.max(axis=1)
        return true_range.rolling(period).mean()

    def _compute_slope(self, series, period=5) -> pd.Series:
        """Compute slope of series over period"""
        return series.diff(period) / period
